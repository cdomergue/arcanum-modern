cmake_minimum_required(VERSION 3.21)
project(WorldEdRenaissance VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ZLIB REQUIRED)
add_library(worlded_formats src/formats.cpp src/objects.cpp src/art.cpp src/installation.cpp src/building.cpp src/world.cpp)
target_include_directories(worlded_formats PUBLIC include)
target_compile_features(worlded_formats PUBLIC cxx_std_17)
target_link_libraries(worlded_formats PRIVATE ZLIB::ZLIB)
if(MSVC)
    target_compile_options(worlded_formats PRIVATE /W4 /utf-8)
else()
    target_compile_options(worlded_formats PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(worlded src/main.cpp)
target_link_libraries(worlded PRIVATE worlded_formats)
if(WIN32)
    target_link_libraries(worlded PRIVATE shell32)
endif()
if(MSVC)
    target_compile_options(worlded PRIVATE /W4 /utf-8)
else()
    target_compile_options(worlded PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME worlded_formats
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_worlded.py" "$<TARGET_FILE:worlded>")
endif()
install(TARGETS worlded RUNTIME DESTINATION bin)

option(WORLDED_BUILD_GUI "Build the Qt resource explorer" OFF)
if(WORLDED_BUILD_GUI)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent)
    add_executable(script-editor src/script_main.cpp)
    target_link_libraries(script-editor PRIVATE worlded_formats Qt6::Widgets)
    if(MSVC)
        target_compile_options(script-editor PRIVATE /utf-8)
    endif()
    install(TARGETS script-editor RUNTIME DESTINATION bin)

    add_executable(worlded-house-export ../house/export-scene.cpp)
    target_link_libraries(worlded-house-export PRIVATE worlded_formats Qt6::Core)
    if(MSVC)
        target_compile_options(worlded-house-export PRIVATE /utf-8)
    endif()
    install(TARGETS worlded-house-export RUNTIME DESTINATION bin)
    add_executable(worlded-explorer src/explorer.cpp src/scene_assets.cpp)
    target_link_libraries(worlded-explorer PRIVATE worlded_formats Qt6::Widgets Qt6::Concurrent)
    if(MSVC)
        target_compile_options(worlded-explorer PRIVATE /W4 /utf-8)
    else()
        target_compile_options(worlded-explorer PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS worlded-explorer RUNTIME DESTINATION bin)
    set(WORLDED_NATIVE_HELPER "" CACHE FILEPATH "Matching engine worlded-native-inventory executable")
    if(WORLDED_NATIVE_HELPER)
        if(NOT EXISTS "${WORLDED_NATIVE_HELPER}")
            message(FATAL_ERROR "WORLDED_NATIVE_HELPER does not exist: ${WORLDED_NATIVE_HELPER}")
        endif()
        add_custom_command(TARGET worlded-explorer POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WORLDED_NATIVE_HELPER}"
                "$<TARGET_FILE_DIR:worlded-explorer>/"
            VERBATIM)
        install(PROGRAMS "${WORLDED_NATIVE_HELPER}" DESTINATION bin)
    endif()
    if(BUILD_TESTING)
        add_test(NAME worlded_house_export COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_house_export.py" "$<TARGET_FILE:worlded-house-export>")
        find_package(Qt6 REQUIRED COMPONENTS Test)
        add_executable(worlded-gui-check test_explorer.cpp src/scene_assets.cpp)
        target_link_libraries(worlded-gui-check PRIVATE worlded_formats Qt6::Widgets Qt6::Concurrent Qt6::Test)
        if(MSVC)
            target_compile_options(worlded-gui-check PRIVATE /utf-8)
            # The integration harness keeps several sector fixtures on its stack.
            target_link_options(worlded-gui-check PRIVATE /STACK:8388608)
        endif()
        add_test(NAME worlded_gui COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_worlded_gui.py" "$<TARGET_FILE:worlded-gui-check>")
    endif()
endif()

if(WORLDED_BUILD_GUI AND BUILD_TESTING)
    add_executable(worlded-campaign-check test_campaign.cpp)
    target_link_libraries(worlded-campaign-check PRIVATE worlded_formats Qt6::Widgets ZLIB::ZLIB)
    if(MSVC)
        target_compile_options(worlded-campaign-check PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(worlded-campaign-check PRIVATE -UNDEBUG)
    endif()
    add_test(NAME worlded_campaign COMMAND worlded-campaign-check -platform offscreen)
endif()

if(WORLDED_BUILD_GUI AND BUILD_TESTING)
    add_executable(worlded-inventory-check test_inventory.cpp src/scene_assets.cpp)
    target_link_libraries(worlded-inventory-check PRIVATE worlded_formats Qt6::Widgets)
    if(MSVC)
        target_compile_options(worlded-inventory-check PRIVATE /utf-8)
    endif()
    add_test(NAME worlded_inventory COMMAND worlded-inventory-check -platform offscreen)
endif()

if(WORLDED_BUILD_GUI)
    target_link_libraries(worlded-explorer PRIVATE ZLIB::ZLIB)
    if(BUILD_TESTING)
        target_link_libraries(worlded-gui-check PRIVATE ZLIB::ZLIB)
    endif()
endif()

if(WORLDED_BUILD_GUI AND BUILD_TESTING)
    add_executable(worlded-update-export-check test_world_update_export.cpp)
    target_link_libraries(worlded-update-export-check PRIVATE worlded_formats Qt6::Core)
    if(MSVC)
        target_compile_options(worlded-update-export-check PRIVATE /utf-8)
    endif()
    add_test(NAME worlded_update_export COMMAND worlded-update-export-check)
endif()

if(WORLDED_BUILD_GUI AND BUILD_TESTING)
    add_executable(script-editor-check test_script_editor.cpp)
    target_link_libraries(script-editor-check PRIVATE worlded_formats Qt6::Widgets)
    target_compile_definitions(script-editor-check PRIVATE SCRIPT_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../examples/scripts")
    if(MSVC)
        target_compile_options(script-editor-check PRIVATE /utf-8)
    endif()
    add_test(NAME script_editor COMMAND script-editor-check -platform offscreen)
    add_test(NAME script_editor_cli COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_script_editor_cli.py" "$<TARGET_FILE:script-editor>")
endif()

# Windows build-tree executables must find the Qt runtime without a shell PATH override.
if(WIN32 AND WORLDED_BUILD_GUI)
    foreach(worlded_program script-editor-check script-editor worlded-explorer worlded-house-export worlded-gui-check
            worlded-campaign-check worlded-inventory-check worlded-update-export-check)
        if(TARGET ${worlded_program})
            add_custom_command(TARGET ${worlded_program} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    $<TARGET_RUNTIME_DLLS:${worlded_program}> $<TARGET_FILE_DIR:${worlded_program}>
                COMMAND_EXPAND_LISTS VERBATIM)
            foreach(qt_platform QWindowsIntegrationPlugin QOffscreenIntegrationPlugin)
                if(TARGET Qt6::${qt_platform})
                    add_custom_command(TARGET ${worlded_program} POST_BUILD
                        COMMAND ${CMAKE_COMMAND} -E make_directory
                            "$<TARGET_FILE_DIR:${worlded_program}>/platforms"
                        COMMAND ${CMAKE_COMMAND} -E copy_if_different
                            $<TARGET_FILE:Qt6::${qt_platform}>
                            "$<TARGET_FILE_DIR:${worlded_program}>/platforms"
                        VERBATIM)
                endif()
            endforeach()
        endif()
    endforeach()
endif()
